local function IsStaffPlayer(p)
    if p.Team and p.Team.Name == "STAFF" then return true end
    if table.find(StaffList, p.Name) then return true end
    local ok, rank = pcall(function() return p:GetRoleInGroup(GrupoSintoniaID) end)
    if ok and rank and rank:lower():find("staff") then return true end
    local ok2, rankN = pcall(function() return p:GetRankInGroup(GrupoSintoniaID) end)
    if ok2 and rankN and rankN > 1 then return true end
    return false
end

local function CheckSafeZone(alvo)
    if not alvo or not alvo.Character then return false end
    if alvo.Character:FindFirstChildOfClass("ForceField") then return true end
    if alvo:GetAttribute("SafeZone") or alvo.Character:GetAttribute("SafeZone") then return true end
    return false
end

local function EstaVisivel(alvo)
    local pecas = Camera:GetPartsObscuringTarget({Camera.CFrame.Position, alvo.Position}, {LocalPlayer.Character, alvo.Parent})
    return #pecas == 0
end

local function GetClosestTarget()
    local target = nil
    local bestScore = math.huge
    local shortestDist = FOVRadius
    local Center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local prioTarget = nil
    local prioScore = math.huge
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and not table.find(IgnoredPlayers, p.Name) and not (IgnorarADM and IsStaffPlayer(p)) then
            if SafezoneCheckAimbot and CheckSafeZone(p) then continue end
            local part = p.Character:FindFirstChild(TargetPart)
            local hum = p.Character:FindFirstChild("Humanoid")
            if part and hum and (not KillCheck or hum.Health > 0) then
                if TeamCheck and p.Team == LocalPlayer.Team then continue end
                local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen then
                    if WallCheck and not EstaVisivel(part) then continue end
                    local magnitude = (Vector2.new(pos.X,pos.Y) - Center).Magnitude
                    if magnitude > FOVRadius then continue end
                    local score = magnitude
                    if PriorizarMenosVida and not PriorizarMaisPerto then
                        score = hum.Health
                    elseif PriorizarMaisPerto and not PriorizarMenosVida then
                        score = (part.Position - Camera.CFrame.Position).Magnitude
                    elseif PriorizarMenosVida and PriorizarMaisPerto then
                        local worldDist = (part.Position - Camera.CFrame.Position).Magnitude
                        score = (hum.Health / math.max(hum.MaxHealth, 1)) * worldDist
                    end
                    if table.find(PriorizarJogadores, p.Name) then
                        if score < prioScore then prioTarget = part; prioScore = score end
                    else
                        if score < bestScore then target = part; bestScore = score; shortestDist = magnitude end
                    end
                end
            end
        end
    end
    return prioTarget or target
end

local function GetSilentTarget(targetPartName)
    targetPartName = targetPartName or TargetPart
    local char = LocalPlayer.Character
    if not char then return nil end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    local Center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local target = nil
    local prioTarget = nil
    local bestScore = math.huge
    local prioScore = math.huge
    for _, p in pairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        if not p.Character then continue end
        if table.find(IgnoredPlayers, p.Name) then continue end
        if IgnorarADM and IsStaffPlayer(p) then continue end
        if TeamCheck and p.Team == LocalPlayer.Team then continue end
        if SafezoneCheckSilentAim and CheckSafeZone(p) then continue end
        local part = p.Character:FindFirstChild(targetPartName)
        local hum = p.Character:FindFirstChild("Humanoid")
        if not part or not hum then continue end
        if KillCheck and hum.Health <= 0 then continue end
        local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
        if not onScreen then continue end
        if WallCheck and not EstaVisivel(part) then continue end
        local magnitude = (Vector2.new(pos.X, pos.Y) - Center).Magnitude
        if magnitude > FOVRadius then continue end
        local score = magnitude
        if PriorizarMenosVida and not PriorizarMaisPerto then
            score = hum.Health
        elseif PriorizarMaisPerto and not PriorizarMenosVida then
            score = (part.Position - Camera.CFrame.Position).Magnitude
        elseif PriorizarMenosVida and PriorizarMaisPerto then
            local worldDist = (part.Position - Camera.CFrame.Position).Magnitude
            score = (hum.Health / math.max(hum.MaxHealth, 1)) * worldDist
        end
        if table.find(PriorizarJogadores, p.Name) then
            if score < prioScore then prioTarget = part; prioScore = score end
        else
            if score < bestScore then target = part; bestScore = score end
        end
    end
    return prioTarget or target
end

local success_sa, result_sa = pcall(function()
    local RS = game:GetService("ReplicatedStorage")
    local OrpheusShared = RS:WaitForChild("OrpheusShared", 10)
    if not OrpheusShared then return end
    local Framework = OrpheusShared:WaitForChild("GunSys", 10)
    and OrpheusShared.GunSys:WaitForChild("Framework", 10)
    if not Framework then return end
    local WRC = Framework:WaitForChild("WeaponRaycastController", 10)
    if not WRC then return end
    local a = require(WRC)
    local b = clonefunction(a.PerformRaycast)
    a.PerformRaycast = newcclosure(function(c, d, e)
        if not SilentAimEnabled then return b(c, d, e) end
        
        if math.random(1, 100) > SilentHitChance then
            return b(c, d, e)
        end
        
        local effectivePart = TargetPart
        if TirosAntesDaCabeca > 0 then
            if TirosDisparados < TirosAntesDaCabeca then
                effectivePart = "HumanoidRootPart"
            else
                effectivePart = "Head"
            end
            TirosDisparados = TirosDisparados + 1
            if TirosDisparados >= 30 then
                TirosDisparados = 0
            end
        end
        
        local h = GetSilentTarget(effectivePart)
        if not h then return b(c, d, e) end
        local f = LocalPlayer.Character
        local g = f and f:FindFirstChild("HumanoidRootPart")
        if not g then return b(c, d, e) end
        local n = c.Config and c.Config.Spread or 0
        return h, h.Position,
        (g.Position - h.Position).Unit,
        (CFrame.new(workspace.CurrentCamera.Focus.Position,
        workspace.CurrentCamera.CFrame.LookVector * 500)
        * CFrame.Angles(
        math.rad(-n + math.random() * (n * 2)),
        math.rad(-n + math.random() * (n * 2)),
        0)).LookVector
    end)
end)

local KillAuraAtiva = false
local DistanciaAura = 100
local TempoEntreTiros = 0.2
local UltimoTiro = 0

RunService.RenderStepped:Connect(function()
    if not KillAuraAtiva then return end
    if tick() - UltimoTiro < TempoEntreTiros then return end

    local meuPersonagem = LocalPlayer.Character
    if not meuPersonagem or not meuPersonagem:FindFirstChild("HumanoidRootPart") then return end
    
    local temArma = meuPersonagem:FindFirstChildWhichIsA("Tool")
    if not temArma then return end

    local minhaPosicao = meuPersonagem.HumanoidRootPart.Position

    for _, jogador in pairs(Players:GetPlayers()) do
        if jogador ~= LocalPlayer and jogador.Character then
            if SafezoneCheckKillaura and CheckSafeZone(jogador) then continue end
            local alvoHumanoid = jogador.Character:FindFirstChild("Humanoid")
            local alvoHead = jogador.Character:FindFirstChild("Head")
            local alvoRoot = jogador.Character:FindFirstChild("HumanoidRootPart")

            if alvoHumanoid and alvoHead and alvoRoot and alvoHumanoid.Health > 0 then
                local distancia = (minhaPosicao - alvoRoot.Position).Magnitude

                if distancia <= DistanciaAura then
                    UltimoTiro = tick() 
                    
                    pcall(function()
                        local Sucesso, Network = pcall(function()
                            return require(ReplicatedStorage:WaitForChild("Modules"):WaitForChild("ModuleRequiring"))("Network")
                        end)
                        if Sucesso and Network then
                            Network:Send("HitDeArma", alvoHumanoid, alvoHead, 50, Vector3.new(0,0,0), 0, 0)
                        else
                            if temArma:FindFirstChild("OrpheusGunServer") and temArma.OrpheusGunServer:FindFirstChild("InflictTarget") then
                                temArma.OrpheusGunServer.InflictTarget:FireServer(alvoHumanoid, alvoHead, 50, Vector3.new(0,0,0), 0, 0)
                            end
                        end
                    end)
                    break 
                end
            end
        end
    end
end)
